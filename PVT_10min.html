// 計時器引用 let timers = { counter: null, responseTimeout: null, isiTimeout: null, testTimeout: null, nextTrialTimeout: null }; // 生成隨機ISI (2-10秒) function generateISI() { return Math.floor(Math.random() * 8000) + 2000; } // 清除所有計時器 function clearAllTimers() { Object.values(timers).forEach(timer => { if (timer) clearInterval(timer); if (timer) clearTimeout(timer); }); Object.keys(timers).forEach(key => timers[key] = null); } // 顯示特定階段 function showStage(stage) { const stages = ['setup', 'ready', 'test', 'results']; stages.forEach(s => { document.getElementById(`${s}-stage`).classList.add('hidden'); }); document.getElementById(`${stage}-stage`).classList.remove('hidden'); } // 開始新的trial function startTrial() { if (appState.stage === 'test' && !appState.testActive) { return; } clearAllTimers(); appState.showCounter = false; appState.message = '注意紅色方框，準備按鈕！'; appState.waitingForResponse = false; updateUI(); const isi = generateISI(); appState.currentISI = isi; timers.isiTimeout = setTimeout(() => { if (appState.stage === 'test' && !appState.testActive) { return; } appState.showCounter = true; '無反應！' : RECORDED_MESSAGE; appState.waitingForResponse = false; const result = { trial: isPractice ? `練習${appState.practiceCount + 1}` : appState.currentTrial + 1, isi: appState.currentISI, result: 'no_response', rt: null, timestamp: new Date().toISOString() }; startTrial(); } else { endTest(); } } } // 進入全螢幕（必須由使用者點擊觸發） function enterFullscreen() { const el = document.documentElement; try { if (el.requestFullscreen) { el.requestFullscreen().catch(() => {}); } else if (el.webkitRequestFullscreen) { el.webkitRequestFullscreen(); } } catch (e) { console.log('無法進入全螢幕:', e); } } // 離開全螢幕 function exitFullscreen() { try { if (document.fullscreenElement) { document.exitFullscreen().catch(() => {}); 開始時間: ${new Date(appState.sessionStartTime).toLocaleString('zh-TW')}

結束時間: ${new Date(appState.sessionEndTime).toLocaleString('zh-TW')}

`; } // 下載結果（內容不變） function downloadResults() { const successResults = appState.results.filter(r => r.result === 'success'); const avgRT = successResults.length > 0 ? (successResults.reduce((sum, r) => sum + r.rt, 0) / successResults.length).toFixed(2) : 'N/A'; let textContent = `reaction time test results\n`; textContent += `========================\n\n`; textContent += `Subject number: ${appState.subjectId}\n`; textContent += `Test start time: ${new Date(appState.sessionStartTime).toLocaleString('zh-TW')}\n`; textContent += `Test end time: ${new Date(appState.sessionEndTime).toLocaleString('zh-TW')}\n`; textContent += `Total trial number: ${appState.results.length}\n`; textContent += `success trial number: ${successResults.length}\n`; textContent += `mean reaction time: ${avgRT} ms\n\n`; textContent += `trial-by-trial results:\n`; textContent += `trial_No. ISI(ms) Response reaction_time(ms)\n`; textContent += `--------------------------------------\n`; appState.results.forEach((result) => { const resultText = result.result === 'success' ?
